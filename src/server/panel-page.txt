    <script>
      const $ = (id) => document.getElementById(id);
      const STORAGE_KEY = "zcode.panel.token";
      let token = localStorage.getItem(STORAGE_KEY) || "";
      let lastCommand = null;
      let lastQuotaAt = 0;
      let logCursor = 0;
      let loggedIn = false;
      let poolNodes = [];
      let poolThreshold = 10;
      let poolSnapshot = [];

      function render(obj) {
        $("raw").textContent = JSON.stringify(obj, null, 2);
      }

      function setConn(state, text) {
        $("dot").className = "dot " + state;
        $("conn").textContent = text;
      }

      async function cmd(payload) {
        let res;
        try {
          res = await fetch("/api/control", {
            method: "POST",
            headers: { "content-type": "application/json", authorization: "Bearer " + token },
            body: JSON.stringify(payload),
          });
        } catch (err) {
          // Server down / restarting — surface it instead of silently dead buttons.
          setConn("bad", "server unreachable");
          return { ok: false, error: "server unreachable (is zcode-proxy running?)" };
        }
        let body;
        try {
          body = await res.json();
        } catch (err) {
          body = { ok: false, error: "invalid_response" };
        }
        lastCommand = { request: payload, status: res.status, response: body };
        render(lastCommand);
        if (res.status === 401) setConn("bad", "unauthorized");
        return body;
      }

      function esc(value) {
        return String(value).replace(/[&<>"']/g, (ch) => (
          { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[ch]
        ));
      }

      /** Upstream timestamps arrive as seconds or milliseconds — accept both. */
      function fmtTs(value) {
        if (value == null) return "—";
        const ms = value < 1e12 ? value * 1000 : value;
        const date = new Date(ms);
        return Number.isNaN(date.getTime()) ? String(value) : date.toLocaleString();
      }

      /** Compact human unit (100M / 3M / 12k); the exact value rides along in the title. */
      function fmtN(n) {
        if (typeof n !== "number" || !isFinite(n)) return "—";
        const a = Math.abs(n);
        const trim = (s) => s.replace(/\.0$/, "");
        if (a >= 1e9) return trim((n / 1e9).toFixed(a >= 1e10 ? 0 : 1)) + "B";
        if (a >= 1e6) return trim((n / 1e6).toFixed(a >= 1e7 ? 0 : 1)) + "M";
        if (a >= 1e4) return Math.round(n / 1e3) + "k";
        return n.toLocaleString();
      }

      /** Bar fill class by remaining share. */
      function barClass(frac) {
        return frac >= 0.5 ? "ok" : frac >= 0.2 ? "warn" : "bad";
      }

      function chip(id, label, value) {
        $(id).innerHTML = esc(label) + " <b>" + esc(value) + "</b>";
      }

      let statusCache = null;

      async function refreshStatus() {
        if (!token) return;
        const r = await cmd({ cmd: "status" });
        if (!r.ok) {
          setConn("bad", r.error || "error");
          return;
        }
        setConn("ok", "connected");
        loggedIn = !!r.loggedIn;
        statusCache = r;
        chip("c-proxy", "proxy", r.proxyPort ? "listening :" + r.proxyPort : "stopped");
        chip("c-default", "defaults", r.provider + " · " + r.plan);
        $("start").disabled = !!r.proxyPort;
        $("stop").disabled = !r.proxyPort;
        renderOauthStatus(r.oauth);
        renderSettingsSummary();
      }

      /** Live OAuth flow feedback: waiting / completed / failed (from `status`). */
      function renderOauthStatus(o) {
        const box = $("oauth-status");
        if (!o) {
          box.innerHTML = "";
          return;
        }
        const when = fmtTs(o.finishedAt ?? o.startedAt);
        if (o.state === "pending") {
          box.innerHTML =
            '<span class="muted">⏳ ' + esc(o.provider) + " login waiting for browser authorization… (started " +
            esc(new Date(o.startedAt).toLocaleTimeString()) + ", times out after 5 min)</span>";
        } else if (o.state === "completed") {
          box.innerHTML =
            '<span class="ok-text">✓ ' + esc(o.provider) + " login completed — account pooled (" + esc(when) + ")</span>";
        } else {
          box.innerHTML =
            '<span class="danger">✗ ' + esc(o.provider) + " login failed (" + esc(when) + "): " + esc(o.error || "unknown error") + "</span>";
        }
      }

      function renderSettingsSummary() {
        const r = statusCache;
        if (!r) return;
        const oauth = poolSnapshot.filter((e) => e.source === "oauth").length;
        const config = poolSnapshot.filter((e) => e.source === "config").length;
        const rows = [
          ["Proxy listen", r.proxyPort ? "http://0.0.0.0:" + r.proxyPort + " (auth key required)" : "stopped"],
          ["Account pool", poolSnapshot.length + " account(s) — " + oauth + " oauth login" + (oauth === 1 ? "" : "s") + ", " + config + " config key" + (config === 1 ? "" : "s")],
          ["Cooldown threshold", poolSnapshot.length ? poolThreshold + " consecutive key errors (429/401/403) per account" : "—"],
          ["Exit nodes", poolNodes.length ? poolNodes.length + " (managed mihomo, one local listener per node)" : "none — accounts connect directly"],
          ["Login store", loggedIn ? "credential present" : "empty"],
        ];
        $("cfg-summary").innerHTML = rows
          .map(([k, v]) => "<dt>" + esc(k) + "</dt><dd>" + esc(v) + "</dd>")
          .join("");
      }

      /** Balance/limit rows for one snapshot slice (whole snapshot, or one pool account). */
      function quotaRows(snap) {
        const parts = [];
        for (const b of snap.balances || []) {
          const remaining = b.remainingUnits;
          const total = b.totalUnits;
          const pct = total ? Math.min(100, Math.max(0, (remaining / total) * 100)) : null;
          parts.push(
            '<div class="qrow">' +
            '<span class="qname" title="' + esc(b.showName) + '">' + esc(b.showName) + "</span>" +
            '<span class="qnum" title="' + esc(remaining) + " / " + esc(total) + '">' +
            fmtN(remaining) + " / " + fmtN(total) + " " + esc(b.unitType || "") + "</span>" +
            (pct === null ? "<span></span>" : '<span class="bar"><i class="' + barClass(pct / 100) + '" style="width:' + pct.toFixed(1) + '%"></i></span>') +
            (b.expiresAt ? '<span class="qexp">expires ' + esc(fmtTs(b.expiresAt)) + "</span>" : "") +
            "</div>",
          );
        }
        const plan = snap.codingPlan;
        if (plan) {
          for (const limit of plan.limits || []) {
            // Upstream `number` is NOT a comparable total (live TIME_LIMIT row:
            // remaining=3894, number=1), so like the CLI/TUI this shows remaining
            // alone. `percentage` is the one self-consistent signal: when present
            // it is a used share (0-100), and the bar is the remaining share
            // derived from it — never an X/Y fabrication.
            const remaining = limit.remaining;
            const pct = limit.percentage;
            const remainingFrac =
              typeof pct === "number" && isFinite(pct) && pct >= 0 && pct <= 100 ? 1 - pct / 100 : null;
            parts.push(
              '<div class="qrow">' +
              '<span class="qname" title="coding-plan ' + esc(limit.type) + '">' + esc(limit.type) + "</span>" +
              '<span class="qnum" title="' + esc(remaining ?? "—") + '">' +
              (typeof remaining === "number"
                ? fmtN(remaining) + (limit.unit ? " " + esc(limit.unit) : "") + " left"
                : "unavailable") + "</span>" +
              (remainingFrac === null
                ? "<span></span>"
                : '<span class="bar"><i class="' + barClass(remainingFrac) + '" style="width:' + (remainingFrac * 100).toFixed(1) + '%"></i></span>') +
              (limit.nextResetTime ? '<span class="qexp">resets ' + esc(fmtTs(limit.nextResetTime)) + "</span>" : "") +
              "</div>",
            );
          }
        } else if (!(snap.balances || []).length) {
          parts.push('<div class="muted" style="font-size:12px">No usage window reported for this account.</div>');
        }
        for (const err of snap.errors || []) {
          parts.push('<div class="err">' + esc(err) + "</div>");
        }
        if (snap.jwt) {
          parts.push('<div class="muted" style="font-size:11px">credential age: ' + esc(snap.jwt.ageHours.toFixed(1)) + " h</div>");
        }
        return parts;
      }

      function renderQuota(snap) {
        let parts;
        if (snap.accounts && snap.accounts.length) {
          // Pool mode: one section per account (the top-level fields are aggregates).
          parts = [];
          for (const a of snap.accounts) {
            parts.push(
              '<div class="qa"><div class="qa-head">' +
              '<span class="name">' + esc(a.label) + "</span>" +
              '<span class="badge">' + esc(a.provider) + "</span>" +
              '<span class="badge">' + esc(a.plan || "coding-plan") + "</span>" +
              (a.codingPlan && a.codingPlan.level ? '<span class="badge accent">' + esc(a.codingPlan.level) + "</span>" : "") +
              "</div>",
            );
            parts = parts.concat(quotaRows(a));
            parts.push("</div>");
          }
        } else {
          parts = quotaRows(snap);
        }
        $("quota-body").innerHTML = parts.join("") || "Empty snapshot.";
      }

      async function loadQuota() {
        if (!token) return;
        const now = Date.now();
        if (now - lastQuotaAt < 60000) {
          $("quota-age").textContent = "please wait " + Math.ceil((60000 - (now - lastQuotaAt)) / 1000) + "s";
          return;
        }
        lastQuotaAt = now;
        $("quota-age").textContent = "loading…";
        const r = await cmd({ cmd: "quota" });
        if (r.ok && r.quota) {
          renderQuota(r.quota);
          $("quota-age").textContent = "updated " + new Date().toLocaleTimeString();
        } else {
          $("quota-age").textContent = "";
          $("quota-body").innerHTML = '<div class="err">' + esc(r.error || "failed") + "</div>";
        }
      }

      function accountRow(e) {
        const editable = e.source === "oauth";
        const state = e.cooling
          ? '<span class="badge bad">cooling</span>'
          : e.current
            ? '<span class="badge accent">active</span>'
            : '<span class="badge">ready</span>';
        const strikes = e.strikes
          ? '<span class="strike">' + e.strikes + "/" + poolThreshold + " strikes</span>"
          : "";
        const nodeTitle = e.proxyUrl ? esc(e.proxyUrl) : "direct connection";
        const nodeOptions =
          '<option value=""' + (e.proxy ? "" : " selected") + ">auto</option>" +
          poolNodes
            .map((n) => '<option value="' + esc(n) + '"' + (e.proxy === n ? " selected" : "") + ">" + esc(n) + "</option>")
            .join("");
        const planSel =
          '<label>plan</label> <select class="compact" data-plan="' + esc(e.id) + '"' + (editable ? "" : " disabled title=\"set plan in config.yaml (pool.accounts)\"") + ">" +
          '<option value="coding-plan"' + (e.plan === "coding-plan" ? " selected" : "") + ">coding</option>" +
          '<option value="start-plan"' + (e.plan === "start-plan" ? " selected" : "") + ">start</option>" +
          "</select>";
        const nodeSel = poolNodes.length
          ? '<label>exit</label> <select class="compact" data-node="' + esc(e.id) + '" title="' + nodeTitle + '"' + (editable ? "" : " disabled") + ">" + nodeOptions + "</select>"
          : '<span class="badge">' + (e.proxyLabel ? esc(e.proxyLabel) : "direct") + "</span>";
        const remove = editable
          ? '<button class="ghost compact" data-remove="' + esc(e.id) + '" title="remove this stored login">✕</button>'
          : "";
        return (
          '<div class="acc">' +
          '<span class="mark">' + (e.current ? "▶" : "·") + "</span>" +
          '<span class="who"><span class="name">' + esc(e.label) + "</span>" +
          '<span class="meta"><span class="badge">' + esc(e.provider) + "</span>" +
          '<span class="badge">' + esc(e.source === "config" ? "config key" : "oauth login") + "</span>" +
          state + strikes +
          (e.proxyLabel ? ' <span class="muted" style="font-size:11px" title="' + nodeTitle + '">via ' + esc(e.proxyLabel) + "</span>" : ' <span class="muted" style="font-size:11px">direct</span>') +
          "</span></span>" +
          '<span class="controls">' + planSel + nodeSel + remove + "</span>" +
          "</div>"
        );
      }

      /** Account-pool card: rotation state, per-account plan/node editors, oauth remove. */
      async function loadPool(force) {
        if (!token) return;
        // Skip re-render while the user is interacting with a per-account control.
        if (!force && document.activeElement && $("pool-body").contains(document.activeElement)) return;
        const r = await cmd({ cmd: "poolStatus" });
        const box = $("pool-body");
        if (!r.ok) {
          box.innerHTML = '<div class="err">' + esc(r.error || "failed") + "</div>";
          return;
        }
        poolNodes = r.nodes || [];
        poolThreshold = r.threshold || 10;
        poolSnapshot = r.pool || [];
        const cooling = poolSnapshot.filter((e) => e.cooling).length;
        chip("c-pool", "accounts", poolSnapshot.length ? poolSnapshot.length + (cooling ? " · " + cooling + " cooling" : " ok") : "0");
        chip("c-exit", "exit nodes", poolNodes.length || "direct");
        renderSettingsSummary();
        if (!poolSnapshot.length) {
          box.innerHTML = '<span class="muted">Pool is empty — log in below or add <code>pool.accounts</code> in config.yaml.</span>';
          return;
        }
        box.innerHTML = poolSnapshot.map(accountRow).join("");
        box.querySelectorAll("[data-remove]").forEach((btn) => {
          btn.onclick = async () => {
            if (!confirm("Remove this stored login from the account pool?")) return;
            await cmd({ cmd: "removeAccount", id: btn.getAttribute("data-remove") });
            await loadPool(true);
            await refreshStatus();
          };
        });
        box.querySelectorAll("[data-plan]").forEach((sel) => {
          sel.onchange = async () => {
            const id = sel.getAttribute("data-plan");
            const res = await cmd({ cmd: "updateAccount", id, plan: sel.value });
            if (!res.ok) {
              $("login-body").innerHTML = '<span class="err">' + esc(res.error) + "</span>";
              await loadPool(true);
            }
          };
        });
        box.querySelectorAll("[data-node]").forEach((sel) => {
          sel.onchange = async () => {
            const id = sel.getAttribute("data-node");
            const res = await cmd({ cmd: "updateAccount", id, proxy: sel.value });
            if (!res.ok) {
              $("login-body").innerHTML = '<span class="err">' + esc(res.error) + "</span>";
            }
            await loadPool(true);
          };
        });
      }

      async function pollLogs() {
        if (!token || !document.getElementById("logs-auto").checked) return;
        const r = await cmd({ cmd: "getLogs", since: logCursor });
        if (!r.ok) return;
        logCursor = r.nextSince;
        if (!r.lines.length) return;
        const box = $("logs");
        const stick = box.scrollTop + box.clientHeight >= box.scrollHeight - 24;
        box.textContent += r.lines.join("\n") + "\n";
        const lines = box.textContent.split("\n");
        if (lines.length > 500) box.textContent = lines.slice(lines.length - 500).join("\n");
        if (stick) box.scrollTop = box.scrollHeight;
      }

      async function startLogin(provider) {
        if (!token) return;
        $("login-body").innerHTML = '<span class="muted">requesting authorize URL…</span>';
        const r = await cmd({ cmd: "startOAuth", provider });
        if (!r.ok) {
          $("login-body").innerHTML = '<span class="danger">' + esc(r.error) + "</span>";
          return;
        }
        $("login-body").innerHTML =
          'Open this URL on any device: <a href="' + esc(r.authorizeUrl) + '" target="_blank" rel="noreferrer">' +
          esc(r.authorizeUrl) + "</a>";
        // Completion feedback is driven by `status.oauth` (renderOauthStatus);
        // the pool card picks the new account up via the disk-sync poll.
      }

      async function guard(label, payload) {
        if (!token) return;
        const r = await cmd(payload);
        $("action-note").textContent = r.ok ? label + ": ok" : label + ": " + r.error;
        await refreshStatus();
        await loadPool(true);
      }

      $("connect").onclick = async () => {
        token = $("token").value.trim();
        localStorage.setItem(STORAGE_KEY, token);
        logCursor = 0;
        $("logs").textContent = "";
        await refreshStatus();
        await loadPool(true);
      };
      $("refresh").onclick = async () => {
        await refreshStatus();
        await loadPool();
      };
      $("quota").onclick = () => loadQuota();
      $("start").onclick = () => guard("start", { cmd: "startProxy" });
      $("stop").onclick = () => guard("stop", { cmd: "stopProxy" });
      $("login").onclick = () => startLogin("zai");
      $("login-bm").onclick = () => startLogin("bigmodel");
      $("logout").onclick = () => {
        if (
          confirm("Log out and delete ALL stored logins? Config-key accounts in config.yaml are kept. The proxy stops if it is running.")
        )
          guard("logout", { cmd: "logout" });
      };
      $("logs-clear").onclick = () => { $("logs").textContent = ""; };

      if (token) {
        $("token").value = token;
        refreshStatus();
        loadPool(true);
        loadQuota();
      } else {
        setConn("bad", "enter token to connect");
      }
      setInterval(refreshStatus, 5000);
      setInterval(() => loadPool(), 5000);
      setInterval(pollLogs, 2000);
      // No quota polling timer on purpose: the billing gateway rate-limits
      // frequent queries (same reason the TUI refreshes on keypress only).
    </script>
  </body>
</html>
